{/* reference:jsdoc */}

Holds nothing and is settled: it never emits. For a branch of a `switchMap` that should follow nothing.

{/* /reference */}

## Example

```tsx
import { counter, createState, empty, switchMap } from 'synstate-next';
import { Optional } from 'ts-std-forge';

const [running, setRunning] = createState(true);

const ticks = running.pipe(
    // While paused, follow nothing: the last tick is kept.
    switchMap((r) => (r ? counter(100) : empty<number>())),
);

await vi.advanceTimersByTimeAsync(250);

setRunning(false);

await vi.advanceTimersByTimeAsync(1000);

assert.deepStrictEqual(ticks.getSnapshot(), Optional.some(2));
```
