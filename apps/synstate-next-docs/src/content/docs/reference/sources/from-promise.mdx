Emits the outcome of `promise` once, as `Result.ok(value)` or `Result.err(error)`, in an update of its own, and settles.

{/* /reference */}

## Marble diagram

<MarbleDiagram slug="from-promise" />

## Example

```tsx
import { fromPromise } from 'synstate-next';
import { Optional, Result } from 'ts-std-forge';

const answer = fromPromise(Promise.resolve(42));

// Nothing until the promise settles.
assert.isTrue(Optional.isNone(answer.getSnapshot()));

await vi.advanceTimersByTimeAsync(0);

assert.deepStrictEqual(answer.getSnapshot(), Optional.some(Result.ok(42)));

const failed = fromPromise<number, Error>(Promise.reject(new Error('down')));

await vi.advanceTimersByTimeAsync(0);

assert.isTrue(
    Result.isErr(Optional.unwrapOr(failed.getSnapshot(), Result.ok(0))),
);
```
