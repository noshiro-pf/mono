{/* reference:jsdoc */}

Creates an event source, holding no value until its first `emit`.

`emit` is a method: call it on the source (`clicks.emit(event)`), and wrap it to pass it on its own, `(event) => { clicks.emit(event); }`. TypeScript rejects a call of it taken off the source.

{/* /reference */}

## Example

```tsx
import { source } from 'synstate-next';
import { Optional } from 'ts-std-forge';

const ticks = source<number>();

// An event source holds no value until its first `emit`.
assert.isTrue(Optional.isNone(ticks.getSnapshot()));

const mut_seen: number[] = [];

ticks.subscribe((n) => {
    mut_seen.push(n);
});

ticks.emit(1);

ticks.emit(1);

assert.deepStrictEqual(mut_seen, [1, 1]);
```
