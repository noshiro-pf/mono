Creates an event source, holding no value until its first `emit`.

{/* /reference */}

## Example

```tsx
import { source } from 'synstate-next';

const ticks = source<number>();

// An event source holds no value until its first `emit`.
assert.isFalse(ticks.hasValue);

const mut_seen: number[] = [];

ticks.subscribe((n) => {
    mut_seen.push(n);
});

ticks.emit(1);

ticks.emit(1);

assert.deepStrictEqual(mut_seen, [1, 1]);
```
