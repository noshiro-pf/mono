{/* jsdoc-description */}
Transforms each value emitted by the source using a mapping function that also receives the emission index.
{/* /jsdoc-description */}

## Marble Diagram

<MarbleDiagram
    inputs={[
        {
            marbles: [
                { t: 10, label: 'A', color: colors.yellow },
                { t: 30, label: 'B', color: colors.orange },
                { t: 60, label: 'C', color: colors.blue },

    ]}

        marbles: [

            { t: 60, label: '2: C', color: colors.blue },
        ],
        completion: 85,
    }}
    operator="map((x, i) => `${i}: ${x}`)"
/>

## Example

```tsx
//  Timeline:
//
//  num$      "A"      "B"      "C"
//  indexed$  "0: A"   "1: B"   "2: C"
//
//  Explanation:
//  - mapWithIndex transforms each value along with its index
//  - Index starts at 0 and increments with each emission

const num$ = source<string>();

const indexed$ = num$.pipe(map((x, i) => `${i}: ${x}`));

const valueHistory: string[] = [];

indexed$.subscribe((s) => {
    valueHistory.push(s);
});

num$.next('A'); // 0: A

num$.next('B'); // 1: B

num$.next('C'); // 2: C

assert.deepStrictEqual(valueHistory, ['0: A', '1: B', '2: C']);
```
